      if to_status != "admitted": return
      round_id = profile.admission_round_id  # FK column, KHÔNG JSONB lookup
      if not round_id: return
      round = await db.execute(
          select(OfferingAdmissionRound)

      ).scalar_one()
      if round.admit_quota is None: return
      # Query qua FK column ix_profile_round_status index (KHÔNG JSONB)
      admitted_count = await db.execute(
          select(func.count())

          .where(AdmissionProfile.status == "admitted")
      ).scalar()
      if admitted_count >= round.admit_quota:
          raise BusinessRuleViolation(
              f"Round {round_id} đã đủ admit_quota ({round.admit_quota})"
          )
