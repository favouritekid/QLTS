  3. Create Helper Function:

  async def _create_admission_milestone_consultation(
      db: AsyncSession,
      lead: models.Lead,
      stage_id: str,
      status_id: str,
      actor: models.User,
      event: str,
      profile_id: int,
  ) -> None:
      """Create system consultation for admission milestone."""

      # Skip if already at terminal state
      if lead.status == "converted":
          return

      consultation = models.Consultation(
          lead_id=lead.id,
          officer_id=actor.id,
          consultation_status_id=status_id,
          consultation_date=datetime.now(timezone.utc),
          method="system",
          notes=f"[Hệ thống] {event} - Profile #{profile_id}",
          duration_minutes=0,
      )
      db.add(consultation)

      # Update lead pipeline (synced via consultation)
      lead.consultation_status_id = status_id
      lead.pipeline_stage_id = stage_id
      sync_lead_status_from_consultation(lead, await db.get(models.ConsultationStatus, status_id))

      log.info(
          "Admission milestone consultation created",
          lead_id=lead.id,
          event=event,
          stage_id=stage_id,
          status_id=status_id,
          profile_id=profile_id,
      )
