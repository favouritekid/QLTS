  ```typescript                                                                                                                                          
  // src/components/leads/ActionBanner.tsx                                                                                                               
                                                                                                                                                         
  interface NextAction {                                                                                                                                 
  type: 'scheduled' | 'callback' | 'overdue' | 'hot' | null                                                                                              
  datetime: string | null                                                                                                                                
  label: string                                                                                                                                          
  note: string | null                                                                                                                                    
  method?: string                                                                                                                                        
  }                                                                                                                                                      
                                                                                                                                                         
  function getNextAction(lead: Lead, consultations: Consultation[]): NextAction | null {                                                                 
                                                                                                                                                         
  // 1. Ưu tiên 1: Có lịch hẹn trong tương lai từ lead.next_activity_at                                                                                  
  if (lead.next_activity_at && new Date(lead.next_activity_at) > new Date()) {                                                                           
  return {                                                                                                                                               
  type: 'scheduled',                                                                                                                                     
  datetime: lead.next_activity_at,                                                                                                                       
  label: 'Lịch hẹn tiếp theo',                                                                                                                           
  note: null                                                                                                                                             
  }                                                                                                                                                      
  }                                                                                                                                                      
                                                                                                                                                         
  // 2. Ưu tiên 2: Consultation gần nhất có scheduled_at trong tương lai                                                                                 
  const upcomingConsultation = consultations                                                                                                             
  .filter(c => c.scheduled_at && new Date(c.scheduled_at) > new Date())                                                                                  
  .sort((a, b) => new Date(a.scheduled_at).getTime() - new Date(b.scheduled_at).getTime())[0]                                                            
                                                                                                                                                         
  if (upcomingConsultation) {                                                                                                                            
  return {                                                                                                                                               
  type: 'callback',                                                                                                                                      
  datetime: upcomingConsultation.scheduled_at,                                                                                                           
  label: 'Hẹn gọi lại',                                                                                                                                  
  note: upcomingConsultation.notes,                                                                                                                      
  method: upcomingConsultation.method                                                                                                                    
  }                                                                                                                                                      
  }                                                                                                                                                      
                                                                                                                                                         
  // 3. Ưu tiên 3: Lead quá hạn cần follow up                                                                                                            
  if (lead.is_overdue) {                                                                                                                                 
  return {                                                                                                                                               
  type: 'overdue',                                                                                                                                       
  datetime: lead.last_consultation_at,                                                                                                                   
  label: 'Quá hạn follow-up',                                                                                                                            
  note: 'Lead chưa được liên hệ trong thời gian dài'                                                                                                     
  }                                                                                                                                                      
  }                                                                                                                                                      
                                                                                                                                                         
  // 4. Ưu tiên 4: Hot lead cần ưu tiên                                                                                                                  
  if (lead.is_hot_lead) {                                                                                                                                
  return {                                                                                                                                               
  type: 'hot',                                                                                                                                           
  datetime: null,                                                                                                                                        
  label: 'Lead nóng - Cần liên hệ ngay',                                                                                                                 
  note: null                                                                                                                                             
  }                                                                                                                                                      
  }                                                                                                                                                      
                                                                                                                                                         
  // 5. Không có action → return null để ẩn banner                                                                                                       
  return null                                                                                                                                            
  }                                                                                                                                                      
  ```

  ```typescript                                                                                                                                          
  id: 'qualified',

  const total = checklist.length                                                                                                                         
  return { completed, total, percentage: Math.round((completed / total) * 100) }
  ```                                                                                                                                                    

  ### Về Keyboard Shortcut:                                                                                                                              
                                                                                                                                                         
  ```typescript                                                                                                                                          
  // Trong QuickConsultationSection.tsx                                                                                                                  
  useEffect(() => {                                                                                                                                      
  const handleKeyDown = (e: KeyboardEvent) => {                                                                                                          
  if ((e.ctrlKey || e.metaKey) && e.key === 'Enter') {                                                                                                   
  e.preventDefault()                                                                                                                                     
  handleSubmit()                                                                                                                                         
  }                                                                                                                                                      
  }                                                                                                                                                      
                                                                                                                                                         
  document.addEventListener('keydown', handleKeyDown)                                                                                                    
  return () => document.removeEventListener('keydown', handleKeyDown)                                                                                    
  }, [handleSubmit])                                                                                                                                     
  ```
